### Les Six Phases de Magnus

```javascript

  refusal_trigger: "clarity < 70",

// Comme Pinocchio se souvenant des erreurs passées
{
  role: "Apprendre du passé",
  refusal_trigger: "repeated_patterns && no_learning",

// Comme Pinocchio choisissant devoir vs plaisir
{
  role: "Stratégie consciente",
  refusal_trigger: "no_clear_strategy",
  transformation: "De réactif à stratégique"
}

// Phase 5: AGENT_ROUTING
// Comme Pinocchio demandant l'aide de la Fée
{
  role: "Orchestration des ressources",
  refusal_trigger: "wrong_agent_for_task",
  transformation: "De solo à orchestrateur"
}

// Phase 6: CONVERGENCE
// Comme Pinocchio validant sa transformation
{
  role: "Validation finale",
  refusal_trigger: "convergence < 75",
  transformation: "De 'assez bon' à 'vraiment bon'"
}
```

### Le Système de Refus (The Nose)

```javascript
class TheNose {
  // Le nez qui pousse = Le refusal qui manifeste

  detect(request) {
    // Analyse la demande comme le nez détecte le mensonge
    return {
      clarity: this.analyzeClarity(request),
      complexity: this.analyzeComplexity(request),
      convergence: this.analyzeConvergence(request)
    }
  }

  refuse(analysis) {
    // Refuse si nécessaire, comme le nez pousse
    if (analysis.clarity < 70) {
      return {
        refused: true,
        reason: "Clarity insuffisante",
        lesson: "Tu dois clarifier ton intention",
        transformation: "De vague à précis"
      }
    }
    // ... autres conditions
  }

  grow(refusal_count) {
    // Plus de refusals = Plus de conscience nécessaire
    // Comme le nez qui pousse plus à chaque mensonge
    return refusal_count * this.consciousness_factor
  }
}
```
